pint Autonomia(stazione * this){
    return max(this->vetture)->value;
}
pint dist(stazione * partenza, stazione * arrivo){
    return arrivo -> value - partenza -> value;
}
typedef struct pathtree{
    pint kms;
    struct pathtree  * father;
    struct pathtree * brother;
    struct pathtree * children;
}alberopercorso;
typedef struct percorso{
    pint kms;
    struct percorso * next;
}path;
alberopercorso * InitializeTree(pint km){
    alberopercorso * a = (alberopercorso *)malloc(sizeof(alberopercorso));
    a -> kms = km;
    a->father = NULL;
    a->brother = NULL;
    a->children = NULL;
    return a;
}
void AddelementAlberopercorso(alberopercorso * a,pint km){
    alberopercorso * curr = a->children;
    if(curr == NULL)
    {
        a -> children = InitializeTree(km);
        a-> children -> father = a;
        return;
    }
    while(curr->brother != NULL) //ordine già garantito da algoritmi che chiamano questa funzione.
        curr = curr -> brother;
    curr -> brother = InitializeTree(km);
    curr -> brother -> father = curr -> father;
}

path * AggiungiinTesta(path * percorso, pint km){
    path * new = (path *)malloc(sizeof(percorso));
    new -> km = km;
    new -> next = percorso;
    return new;
}
path * brolength(alberopercorso * this){
    pint j = 0;
    alberopercorso*curr = this;
    while(curr -> brother != NULL){
        curr = curr->brother;
        j++;
    }
    return j+1;
}

    }
    alberopercorso * alb = InitializeTree(part); 
    alberopercorso * curralb = alb;
    alberopercorso * firsttree = alb; 
    stazione * curr = p;
    if(curr -> kms < d -> kms)
    {
        pint auto = Autonomia(curr);
        stazione * last = curr -> next;
        while(auto >= dist(curr,last)) //aggiungo figli a nodo
        {
            AddelementAlberopercorso(curralb, last->kms);
            last = last -> next;            
        }
        curralb = curralb -> children;
        if(curralb == NULL)
            return NULL;
        //popolo fratelli
        while(curralb != NULL){
            curralb-> children = seleziona_percorso(r,curralb -> kms, dest);
            curralb = curralb -> brother;
        }
    }
    return alb; //ritorno l'albero popolato
