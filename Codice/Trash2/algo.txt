pint Autonomia(stazione * this){
    return max(this->vetture)->value;
}
pint dist(stazione * partenza, stazione * arrivo){
    return arrivo -> value - partenza -> value;
}
typedef struct percorso{
    pint km;    
    struct percorso * next;
}path;
path * AggiungiinTesta(path * percorso, stazione * stat){
    path * new = (path *)malloc(sizeof(percorso));
    new -> km = stat -> kms;
    new -> next = percorso;
    return new;
}
void AggiungiinCoda(path * percorso, stazione * stat){
    path * new = (path *)malloc(sizeof(percorso));
    new -> km = stat -> kms;
    new -> next = NULL;
    percorso -> next = new;
}
path * percorso_migliore_dritto(route* r, pint p, pint d){
    stazione * part = cerca(r, p);
    stazione * dest = cerca(r, d);
    if(part == NULL || dest == NULL)
        return NULL;
    path * strada = (path *)malloc(sizeof(percorso));
    stazione * curr = dest;
    stazione * last = dest;
    strada -> km = dest -> kms;
    strada -> next = NULL;
    do{
        do{
            curr = curr ->prev;
        }while(curr -> value > part -> value && Autonomia(curr)>=dist(curr, last));
    if(Autonomia(curr->next) >= dist(curr->next, last))
    {
        strada = AggiungiinTesta(strada, curr->next);
        last = curr->next;
    }
    }while(curr -> value > part -> value);
    return strada;
}
path * percorso_migliore_reverso(route* r, pint p, pint d){
    stazione * part = cerca(r, p);
    stazione * dest = cerca(r, d);
    if(part == NULL || dest == NULL)
        return NULL;
    path * strada = (path *)malloc(sizeof(percorso));
    stazione * curr = part;
    stazione * last = part;
    strada -> km = part -> kms;
    strada -> next = NULL;
    do{
        do{
            curr = curr ->next;
        }while(curr->value < dest -> value && Autonomia(last)>=dist(last, curr));
        if(Autonomia(last)>= dist(last, curr -> prev))
        { 
            AggiungiinCoda(strada, curr->prev);
            last = curr ->prev;
        }
    } while(curr->value < dest->value);
    return strada;
}
