void inserimento_testa(route * r, pint km, stazione * ref, pint cell){
    stazione * new = EXNOVOstation(km);
    if(ref == r->AUTOSTRADA[cell])
    {
        new ->next = ref;
        new ->prev = ref ->prev;
        if(r->AUTOSTRADA[cell] != NULL)
            ref ->prev = new;
        r->AUTOSTRADA[cell] = new;
        //fase scrolling-checking
        scrolling_forward(r,cell,lastline(r));
        r->lastindex++;
        Check(r);
    }
}

void scrollascrollascrolla(route * r, pint stop)
{
    pint nodes = autolen(r);
    pint periodi = (pint)(stop/nodes);
    pint i,j,k;
    pint pos, per, lastcont;
    for(i=0; i<periodi; i++){
        per = i*nodes;
        for(j=0; j < nodes; j++)
            for(k=0; k < j; k++){
                pos = per + j; //i-esimo periodo, nodesima riga.
                r->AUTOSTRADA[pos] = r->AUTOSTRADA[pos]->next;
            }
    }
    //ultimo periodo
    lastcont = 0;
    for(j=pos; j < stop; j++){
        for(k=0; k < lastcont; k++)
            r->AUTOSTRADA[j] = r->AUTOSTRADA[j]->next;
        lastcont++;
    }

}
